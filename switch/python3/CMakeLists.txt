cmake_minimum_required(VERSION 3.18)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "$ENV{DEVKITPRO}/libounce/cmake/Switch2.cmake" CACHE FILEPATH "Switch 2 toolchain")
endif()
project(renounce_python3_probe C)

set(RENONCE_PYTHON3_INCLUDE "" CACHE PATH "Cross-compiled Python 3 include directory")
set(RENONCE_PYTHON3_ROMFS "" CACHE PATH "Python 3-only romfs directory")
set(RENONCE_PYTHON3_CONFIG_INCLUDE "" CACHE PATH "Directory containing target pyconfig.h")
set(RENONCE_PYTHON3_ARCHIVE "" CACHE FILEPATH "Cross-compiled static Python 3 archive")
set(RENONCE_EXTENSION_ARCHIVE "" CACHE FILEPATH "Ren'Py 8 extension archive")
set(RENONCE_MODULE_TABLE "" CACHE FILEPATH "Generated static module registration C file")
if(NOT EXISTS "${RENONCE_PYTHON3_INCLUDE}/Python.h" OR
   NOT EXISTS "${RENONCE_PYTHON3_CONFIG_INCLUDE}/pyconfig.h" OR
   NOT EXISTS "${RENONCE_PYTHON3_ROMFS}/Contents/lib.zip" OR
   NOT EXISTS "${RENONCE_PYTHON3_ARCHIVE}")
    message(FATAL_ERROR "Provide RENONCE_PYTHON3_INCLUDE and RENONCE_PYTHON3_ARCHIVE from the same Python 3 cross build")
endif()

add_executable(renounce-python3 launcher.c)
option(RENONCE_ENGINE_BOOT "Start the full Ren'Py 8.4.1 engine" OFF)
option(RENONCE_RELEASE "Build without Python bootstrap diagnostics" OFF)
set(RENONCE_GAME_ID "renpy-8.4.1-engine" CACHE STRING "Per-NRO data folder below renounce beside the NRO")
if(NOT RENONCE_GAME_ID MATCHES "^[A-Za-z0-9._-]+$")
    message(FATAL_ERROR "RENONCE_GAME_ID must use letters, digits, dot, underscore or hyphen")
endif()
target_compile_definitions(renounce-python3 PRIVATE RENOUNCE_GAME_ID="${RENONCE_GAME_ID}")
if(RENONCE_ENGINE_BOOT)
    target_compile_definitions(renounce-python3 PRIVATE RENOUNCE_ENGINE_BOOT=1)
endif()
if(RENONCE_RELEASE)
    target_compile_definitions(renounce-python3 PRIVATE RENOUNCE_RELEASE=1)
endif()
target_include_directories(renounce-python3 PRIVATE
    "${RENONCE_PYTHON3_INCLUDE}" "${RENONCE_PYTHON3_CONFIG_INCLUDE}" "$ENV{DEVKITPRO}/libounce/include")
target_link_libraries(renounce-python3 "${RENONCE_PYTHON3_ARCHIVE}" m)
if(RENONCE_EXTENSION_ARCHIVE)
    if(NOT EXISTS "${RENONCE_EXTENSION_ARCHIVE}" OR NOT EXISTS "${RENONCE_MODULE_TABLE}")
        message(FATAL_ERROR "Provide extension archive and generated module table")
    endif()
    target_sources(renounce-python3 PRIVATE "${RENONCE_MODULE_TABLE}")
    target_compile_definitions(renounce-python3 PRIVATE RENOUNCE_LINK_EXTENSIONS=1)
    find_package(PkgConfig REQUIRED)
    foreach(pkg IN ITEMS sdl2 SDL2_image SDL2_mixer SDL2_ttf freetype2
            libavformat libavcodec libavutil libswresample libswscale
            fribidi zlib egl glesv2 glapi libdrm_nouveau)
        string(MAKE_C_IDENTIFIER "${pkg}" pkg_id)
        pkg_check_modules(${pkg_id} REQUIRED IMPORTED_TARGET "${pkg}")
        list(APPEND extension_libs "PkgConfig::${pkg_id}")
    endforeach()
    # Switch 1 portlibs export absolute libnx and Mesa paths. Never mix their
    # runtime/graphics archives with libounce and OunceVK in a Switch 2 NRO.
    foreach(pkg_target IN LISTS extension_libs)
        get_target_property(pkg_links ${pkg_target} INTERFACE_LINK_LIBRARIES)
        if(pkg_links)
            set(safe_links "")
            foreach(lib IN LISTS pkg_links)
                if(lib STREQUAL "$ENV{DEVKITPRO}/libnx/lib/libnx.a")
                    set(lib "$ENV{DEVKITPRO}/libounce/lib/libnx.a")
                elseif(lib STREQUAL "$ENV{DEVKITPRO}/portlibs/switch/lib/libEGL.a")
                    set(lib "$ENV{DEVKITPRO}/portlibs/switch2/lib/libEGL.a")
                elseif(lib STREQUAL "$ENV{DEVKITPRO}/portlibs/switch/lib/libglapi.a")
                    set(lib "$ENV{DEVKITPRO}/portlibs/switch2/lib/libglapi.a")
                elseif(lib STREQUAL "$ENV{DEVKITPRO}/portlibs/switch/lib/libdrm_nouveau.a")
                    set(lib "$ENV{DEVKITPRO}/portlibs/switch2/lib/libdrm_nouveau.a")
                endif()
                list(APPEND safe_links "${lib}")
            endforeach()
            set_target_properties(${pkg_target} PROPERTIES INTERFACE_LINK_LIBRARIES "${safe_links}")
        endif()
        get_target_property(pkg_includes ${pkg_target} INTERFACE_INCLUDE_DIRECTORIES)
        if(pkg_includes)
            list(REMOVE_ITEM pkg_includes "$ENV{DEVKITPRO}/libnx/include")
            set_target_properties(${pkg_target} PROPERTIES INTERFACE_INCLUDE_DIRECTORIES "${pkg_includes}")
        endif()
        get_target_property(pkg_options ${pkg_target} INTERFACE_COMPILE_OPTIONS)
        if(pkg_options)
            list(FILTER pkg_options EXCLUDE REGEX "libnx/include")
            set_target_properties(${pkg_target} PROPERTIES INTERFACE_COMPILE_OPTIONS "${pkg_options}")
        endif()
    endforeach()
    target_link_libraries(renounce-python3
        -Wl,--start-group "${RENONCE_EXTENSION_ARCHIVE}" ${extension_libs}
        "${CMAKE_SOURCE_DIR}/../../build-renpy8/8.4.1/assimp/build-5.2.5/lib/libassimp.a"
        "$ENV{DEVKITPRO}/portlibs/switch/lib/libharfbuzz.a"
        -Wl,--end-group stdc++)
endif()
if(RENONCE_RELEASE)
    nx_generate_nacp(renounce-python3.nacp NAME "renounce Ren'Py 8.4.1"
        AUTHOR "renounce" VERSION "8.4.1")
else()
    nx_generate_nacp(renounce-python3.nacp NAME "renOunce Python 3 probe"
        AUTHOR "renOunce" VERSION "0.1")
endif()
option(RENONCE_PACKAGE_PROBE "Package the probe as an executable NRO" OFF)
if(RENONCE_PACKAGE_PROBE)
    nx_create_nro(renounce-python3 NACP renounce-python3.nacp
        ROMFS "${RENONCE_PYTHON3_ROMFS}")
endif()
